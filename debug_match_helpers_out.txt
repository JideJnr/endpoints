123:     """Convert fractional odds (``"5/2"``) to implied probability."""
124:     if not value or "/" not in str(value):
125:         return None
126:     top, bottom = str(value).split("/", 1)
127:     numerator = _to_float(top)
128:     denominator = _to_float(bottom)
129:     if numerator is None or denominator in (None, 0):
130:         return None
131:     decimal = numerator / denominator + 1
